cmake_minimum_required(VERSION 3.13)
project(ClearCastReceiver LANGUAGES C CXX)

if(NOT WIN32 OR NOT DEFINED UXPLAY_SOURCE_DIR)
  message(FATAL_ERROR "A Windows toolchain and UXPLAY_SOURCE_DIR are required.")
endif()

add_subdirectory("${UXPLAY_SOURCE_DIR}" uxplay EXCLUDE_FROM_ALL)
add_library(clearcast_receiver SHARED "${UXPLAY_SOURCE_DIR}/uxplay.cpp")
target_compile_features(clearcast_receiver PRIVATE cxx_std_11)
foreach(target clearcast_receiver airplay renderers)
  target_compile_definitions(${target} PRIVATE CLEARCAST_IN_PROCESS)
endforeach()
target_include_directories(clearcast_receiver PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${UXPLAY_SOURCE_DIR}")
target_include_directories(renderers PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(clearcast_receiver PRIVATE renderers airplay)
set_target_properties(clearcast_receiver PROPERTIES PREFIX "")

include(CTest)
if(BUILD_TESTING)
  add_executable(clearcast_http_headers_test tests/http_headers_test.c "${UXPLAY_SOURCE_DIR}/lib/http_request.c")
  target_include_directories(clearcast_http_headers_test PRIVATE "${UXPLAY_SOURCE_DIR}/lib")
  target_link_libraries(clearcast_http_headers_test PRIVATE llhttp)
  add_test(NAME http_headers_case_insensitive COMMAND clearcast_http_headers_test)
  add_executable(clearcast_http_security_test tests/http_request_security_test.c "${UXPLAY_SOURCE_DIR}/lib/http_request.c")
  target_include_directories(clearcast_http_security_test PRIVATE "${UXPLAY_SOURCE_DIR}/lib")
  target_link_libraries(clearcast_http_security_test PRIVATE llhttp)
  add_test(NAME http_request_security COMMAND clearcast_http_security_test)
  add_executable(clearcast_video_context_test tests/video_context_test.c)
  target_include_directories(clearcast_video_context_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(clearcast_video_context_test PRIVATE renderers)
  add_test(NAME video_context_lifecycle COMMAND clearcast_video_context_test)
endif()
